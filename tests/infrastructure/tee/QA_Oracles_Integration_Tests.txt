curl -X 'POST' \
  'http://localhost:8000/api/v1/oracle/validate-initiative' \
  -H 'accept: application/json' \
  -H 'Content-Type: application/json' \
  -d '{
  "author_did": "did:transceptor:mentor:0xCOMPROMISED",
  "initiative_id_hash": "0xINITIATIVE_FRAUD_HASH_666",
  "raw_text": "Программа развития когнитивных треков кампуса Агоры на следующий образовательный спринт.",
